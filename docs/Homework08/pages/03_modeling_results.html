Split Data

set.seed(123)
bike_split <- initial_split(daily_bike, prop = 0.75, strata = Seasons)
bike_train <- training(bike_split)
bike_test  <- testing(bike_split)

Recipe 1: base model

# This recipe prepares the data for modeling. 
# It creates a weekday and weekend variable from the date, removes the original Date column,
# standardizes all numeric predictors so they are on the same scale,
# and converts all categorical variables into dummy (0/1) variables.
# These steps help the linear model interpret the data correctly.


recipe_1 <- recipe(Rented_Bike_sum ~ ., data = bike_train) |>
step_mutate(day_type = factor(if_else(wday(Date) %in% c(1,7), "weekend", "weekday"))) |>
step_rm(Date) |>
step_normalize(all_numeric_predictors()) |>
step_dummy(all_nominal_predictors())

Recipe 2: add interactions

# This recipe builds on the first one. It includes weekday and weekend,
# removes the Date column, standardizes numeric variables,
# and makes dummy variables for categorical features.
# It also adds interaction terms so the model can capture how
# seasons and holidays work together, how seasons and temperature interact,
# and how temperature and rainfall interact. This helps the model learn
# more complex relationships in the data.


recipe_2 <- recipe(Rented_Bike_sum ~ ., data = bike_train) |>
step_mutate(day_type = factor(if_else(wday(Date) %in% c(1,7), "weekend", "weekday"))) |>
step_rm(Date) |>
step_normalize(all_numeric_predictors()) |>
step_dummy(all_nominal_predictors()) |>
step_interact(
terms = ~ starts_with("Seasons") * starts_with("Holiday") +
starts_with("Seasons") * Temperature_mean +
Temperature_mean * Rainfall_sum
)

Recipe 3: add quadratic terms (polynomials)

# This recipe includes everything from recipe 2 (weekday and weekend,
# removing Date, standardizing numerics, dummy variables, and interactions).
# It also adds polynomial (squared) terms for several numeric predictors.
# These quadratic terms help the model capture curved relationships,
# such as rentals increasing up to a point and then leveling off.
# This makes the model more flexible for weather related effects.


recipe_3 <- recipe(Rented_Bike_sum ~ ., data = bike_train) |>
step_mutate(day_type = factor(if_else(wday(Date) %in% c(1,7), "weekend", "weekday"))) |>
step_rm(Date) |>
step_normalize(all_numeric_predictors()) |>
step_dummy(all_nominal_predictors()) |>
step_interact(
terms = ~ starts_with("Seasons") * starts_with("Holiday") +
starts_with("Seasons") * Temperature_mean +
Temperature_mean * Rainfall_sum
) |>
step_poly(
Temperature_mean,
Humidity_mean,
Windspeed_mean,
Visibility_mean,
Dew_Point_Temperature_mean,
Solar_Radiation_mean,
degree = 2
)

Model Setup

# Here I set up our linear regression model and combine it with each recipe.
# I create one workflow for each recipe so we can compare the three models:
# Model 1 uses the basic recipe, Model 2 adds interaction terms,
# and Model 3 adds both interactions and polynomial terms.



bike_mod <- linear_reg() |>
set_engine("lm")

bike_wfl_1 <- workflow() |>
add_recipe(recipe_1) |>
add_model(bike_mod)

bike_wfl_2 <- workflow() |>
add_recipe(recipe_2) |>
add_model(bike_mod)

bike_wfl_3 <- workflow() |>
add_recipe(recipe_3) |>
add_model(bike_mod)

Cross-Validation

# I use 10 fold cross validation to compare the performance of the three models.
# The training data is split into 10 parts, and each model is tested across all folds.
# Measure error using RMSE, which tells us how far the predictions are from the actual rentals.


set.seed(42)
bike_10_fold <- vfold_cv(bike_train, v = 10)

bike_res_1 <- fit_resamples(
bike_wfl_1,
resamples = bike_10_fold,
metrics = metric_set(rmse)
)
